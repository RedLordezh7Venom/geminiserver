Below is a **complete, ready‑to‑copy** example of a modern, responsive web‑page that contains a clean, accessible form. It uses only HTML 5, CSS 3 (no external frameworks) and a tiny bit of vanilla JavaScript for client‑side validation and a friendly “thank‑you” modal. const modal = document.getElementById('thankYouModal'); const closeBtn = document.getElementById('closeModal'); // Helper to show error messages const showError = (input, message) => { const formGroup = input.parentElement; const errorEl = formGroup.querySelector('.error-msg'); errorEl.textContent = message; errorEl.style.display = 'block'; input.classList.add('invalid'); }; const clearError = (input) => { const formGroup = input.parentElement; const errorEl = formGroup.querySelector('.error-msg'); errorEl.textContent = ''; errorEl.style.display = 'none'; input.classList.remove('invalid'); }; // Basic validation rules const validators = { fullName: (val) => val.trim().length >= 2 || 'Please enter a valid name.', email: (val) => /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(val) || 'Enter a valid email address.', phone: (val) => { if (!val) return true; // optional return /^\+?[0-9\s\-\(\)]{7,20}$/.test(val) || 'Enter a valid phone number.'; }, subject: (val) => val !== '' || 'Please select a subject.', message: (val) => val.trim().length >= 10 || 'Message should be at least 10 characters.' }; // Validate a single field const validateField = (input) => { const name = input.name; const validator = validators[name]; if (!validator) return true; // no rule defined const result = validator(input.value); if (result === true) { clearError(input); return true; } else { showError(input, result); return false; } }; // Attach blur listeners for instant feedback [...form.elements].forEach(el => { if (el.tagName.toLowerCase() !== 'button') { el.addEventListener('blur', () => validateField(el)); } }); // Form submit handler form.addEventListener('submit', (e) => { e.preventDefault(); // Run validation on every field const isValid = [...form.elements].every(el => { if (el.tagName.toLowerCase() === 'button') return true; return validateField(el); }); if (!isValid) { // Focus first invalid field const firstInvalid = form.querySelector('.invalid'); firstInvalid && firstInvalid.focus(); return; } // ---- At this point you could send the data via fetch/AJAX ---- // For demo purposes we just show the modal. modal.classList.remove('hidden'); form.reset(); }); // Close modal closeBtn.addEventListener('click', () => modal.classList.add('hidden')); // Close modal on ESC key document.addEventListener('keydown', (e) => { if (e.key === 'Escape' && !modal.classList.contains('hidden')) { modal.classList.add('hidden'); } }); }); ``` --- ## 🎉 How it works | Piece | What it does | |------|--------------| | **HTML** | Semantic markup (`
